USER {
        int user_id PK
        string username
        string email
        date inscription_date
        int total_points
    }

    SPORT_CATEGORY {
        int cat_id PK
        string nom
        int parent_cat_id FK
    }

    SESSION {
        int session_id PK
        datetime debut
        datetime fin
        float distance
        float denivele
        float vitesse_moyenne
        float vitesse_max
    }

    CHALLENGE {
        int defi_id PK
        int createur_id FK
        string titre
        text description
        date date_debut
        date date_fin
        string type_defi (individuel | collectif)
        string objectif_type (distance | denivele | temps | jours_actifs)
        float valeur_cible
        int capacite_max "null pour indiv"
    }

    --tables d'association
    PARTICIPATION {
        int participation_id PK
        int user_id FK
        int defi_id FK
        float progression
        boolean termine
    }

    INVITATION {
        int invitation_id PK
        int inviteur_id FK
        int invite_id FK
        int defi_id FK
        string statut (en_attente | acceptee | refusee)
        date created_at
    }

    FOLLOW {
        int follower_id PK,FK
        int followed_id PK,FK
    }

    USER_BADGE {
        int user_id PK,FK
        int badge_id PK,FK
        date date_obtention
    }
